#include <iostream>
#include <string>

int main(int argc, char* argv[]) {
    std::string nome;
    int idade = 0;
    std::cout << "Qual é o seu nome? ";
    std::getline(std::cin, nome);
    std::cout << "Idade: ";
    std::cin >> idade;
    std::cout << "Oi, " << nome << "! Daqui a 10 anos você terá " << idade + 10 << ".\n";
    std::cout << "Você passou " << argc - 1 << " argumentos.\n";
    return 3;
}
